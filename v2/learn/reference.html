Python reference

Every method of strings, lists, tuples, sets and dictionaries, plus the built-in functions. Each example was run in real Python.

String methods

Strings never change, so every method that edits text gives you a new string. Store it: s = s.strip().

MethodWhat it doesExampleOutput
capitalize()Capital first letter, the rest small
print("hELLO world".capitalize())
Hello world
casefold()All small letters, for comparing text
print("PYTHON".casefold() == "python")
True
center(width, fill)Centres the text in a given width
print("hi".center(8, "*"))
***hi***
count(x)How many times x appears
print("banana".count("a"))
3
endswith(x)True if the text ends with x
print("report.csv".endswith(".csv"))
True
find(x)Position of x, or -1 if missing
print("data".find("t"), "data".find("z"))
2 -1
format(...)Fills {} slots (f-strings are easier)
print("{} is {} years".format("Asha", 25))
Asha is 25 years
index(x)Position of x; ValueError if missing
print("data".index("t"))
2
isalnum()True if only letters and digits
print("abc123".isalnum(), "abc 123".isalnum())
True False
isalpha()True if only letters
print("Python".isalpha(), "Py3".isalpha())
True False
isdigit()True if only digits
print("2024".isdigit(), "20.5".isdigit())
True False
islower()True if all letters are small
print("hello".islower())
True
isspace()True if only spaces, tabs or new lines
print("   ".isspace())
True
istitle()True if Every Word Starts Capital
print("Hello World".istitle())
True
isupper()True if all letters are capital
print("NASA".isupper())
True
join(items)Glues a list of strings with this text between
print("-".join(["2026", "10", "28"]))
2026-10-28
ljust(width, fill)Pads on the right to a given width
print("hi".ljust(5, ".") + "|")
hi...|
lower()All small letters
print("Hello".lower())
hello
lstrip()Removes spaces on the left only
print("[" + "  hi  ".lstrip() + "]")
[hi  ]
partition(sep)Splits once into (before, sep, after)
print("name=Asha".partition("="))
('name', '=', 'Asha')
removeprefix(x)Removes x from the start, if it is there
print("INV-204".removeprefix("INV-"))
204
removesuffix(x)Removes x from the end, if it is there
print("sales.csv".removesuffix(".csv"))
sales
replace(old, new)Changes every old to new
print("1,200,000".replace(",", ""))
1200000
rfind(x)Like find(), but searches from the right
print("a/b/c".rfind("/"))
3
rjust(width, fill)Pads on the left to a given width
print("42".rjust(5, "0"))
00042
rsplit(sep, n)Splits from the right, at most n times
print("a.b.c".rsplit(".", 1))
['a.b', 'c']
rstrip()Removes spaces on the right only
print("[" + "  hi  ".rstrip() + "]")
[  hi]
split(sep)Breaks text into a list
print("Asha,82,Pune".split(","))
['Asha', '82', 'Pune']
splitlines()Breaks text into a list of lines
print("line 1\nline 2".splitlines())
['line 1', 'line 2']
startswith(x)True if the text starts with x
print("+91 98480".startswith("+91"))
True
strip()Removes spaces at both ends
print("[" + "  hi  ".strip() + "]")
[hi]
swapcase()Small becomes capital and capital becomes small
print("PyThOn".swapcase())
pYtHoN
title()Capital first letter of every word
print("asha reddy".title())
Asha Reddy
upper()All capital letters
print("pune".upper())
PUNE
zfill(width)Pads with zeros on the left
print("7".zfill(3))
007

List methods

Most list methods change the list itself and give back None. Do not write a = a.sort().

MethodWhat it doesExampleOutput
append(x)Adds one item at the end
a = [1, 2]
a.append(3)
print(a)
[1, 2, 3]
clear()Removes every item
a = [1, 2, 3]
a.clear()
print(a)
[]
copy()A separate copy of the list
a = [1, 2]
b = a.copy()
b.append(9)
print(a, b)
[1, 2] [1, 2, 9]
count(x)How many times x appears
print([1, 2, 2, 3, 2].count(2))
3
extend(items)Adds each item of another list
a = [1, 2]
a.extend([3, 4])
print(a)
[1, 2, 3, 4]
index(x)Position of the first x; ValueError if missing
print(["a", "b", "c"].index("c"))
2
insert(i, x)Puts x at position i
a = ["b", "c"]
a.insert(0, "a")
print(a)
['a', 'b', 'c']
pop(i)Removes and gives back item i (default: the last)
a = [10, 20, 30]
x = a.pop()
print(x, a)
30 [10, 20]
remove(x)Removes the first x; ValueError if missing
a = ["tea", "milk", "tea"]
a.remove("tea")
print(a)
['milk', 'tea']
reverse()Reverses the list in place
a = [1, 2, 3]
a.reverse()
print(a)
[3, 2, 1]
sort()Sorts the list in place
a = [3, 1, 2]
a.sort(reverse=True)
print(a)
[3, 2, 1]

Tuple methods

Tuples cannot change, so they have only two methods.

MethodWhat it doesExampleOutput
count(x)How many times x appears
print((1, 2, 2, 3).count(2))
2
index(x)Position of the first x
print(("a", "b", "c").index("b"))
1

Set methods

Sets keep only unique items and have no order. Operators like | and & do the same jobs as several methods.

MethodWhat it doesExampleOutput
add(x)Adds one item
s = {1, 2}
s.add(3)
print(s)
{1, 2, 3}
clear()Removes every item
s = {1, 2}
s.clear()
print(s)
set()
copy()A separate copy of the set
s = {1, 2}
t = s.copy()
t.add(9)
print(s, t)
{1, 2} {1, 2, 9}
difference(t)Items in s but not in t (same as s - t)
print({1, 2, 3}.difference({2}))
{1, 3}
difference_update(t)Removes t's items from s itself
s = {1, 2, 3}
s.difference_update({2})
print(s)
{1, 3}
discard(x)Removes x; does nothing if missing
s = {1, 2}
s.discard(9)
print(s)
{1, 2}
intersection(t)Items in both (same as s & t)
print({1, 2, 3}.intersection({2, 3, 4}))
{2, 3}
intersection_update(t)Keeps only items also in t
s = {1, 2, 3}
s.intersection_update({2, 3, 4})
print(s)
{2, 3}
isdisjoint(t)True if no items are shared
print({1, 2}.isdisjoint({3, 4}))
True
issubset(t)True if every item is also in t (s <= t)
print({1, 2}.issubset({1, 2, 3}))
True
issuperset(t)True if s contains every item of t (s >= t)
print({1, 2, 3}.issuperset({1, 2}))
True
pop()Removes and gives back some item
s = {7}
print(s.pop(), s)
7 set()
remove(x)Removes x; KeyError if missing
s = {1, 2}
s.remove(2)
print(s)
{1}
symmetric_difference(t)Items in one set but not both (s ^ t)
print({1, 2, 3}.symmetric_difference({3, 4}))
{1, 2, 4}
union(t)Everything from both (same as s | t)
print({1, 2}.union({2, 3}))
{1, 2, 3}
update(t)Adds every item of t to s
s = {1, 2}
s.update([2, 3, 4])
print(s)
{1, 2, 3, 4}

Dictionary methods

A dictionary maps unique keys to values. d[key] raises KeyError for a missing key; get() does not.

MethodWhat it doesExampleOutput
clear()Removes every pair
d = {"a": 1}
d.clear()
print(d)
{}
copy()A separate copy of the dictionary
d = {"a": 1}
e = d.copy()
e["b"] = 2
print(d, e)
{'a': 1} {'a': 1, 'b': 2}
fromkeys(keys, v)New dict with every key set to v
print(dict.fromkeys(["tea", "coffee"], 0))
{'tea': 0, 'coffee': 0}
get(key, default)The value, or default if the key is missing
d = {"tea": 20}
print(d.get("tea"), d.get("milk", 0))
20 0
items()All (key, value) pairs
d = {"tea": 20, "coffee": 45}
print(list(d.items()))
[('tea', 20), ('coffee', 45)]
keys()All keys
d = {"tea": 20, "coffee": 45}
print(list(d.keys()))
['tea', 'coffee']
pop(key)Removes the key and gives back its value
d = {"tea": 20, "coffee": 45}
print(d.pop("tea"), d)
20 {'coffee': 45}
popitem()Removes and gives back the last pair added
d = {"tea": 20, "coffee": 45}
print(d.popitem(), d)
('coffee', 45) {'tea': 20}
setdefault(key, v)The value; adds key with v first if missing
d = {}
d.setdefault("Pune", []).append("Asha")
print(d)
{'Pune': ['Asha']}
update(other)Adds or overwrites pairs from another dict
d = {"tea": 20}
d.update({"tea": 25, "milk": 15})
print(d)
{'tea': 25, 'milk': 15}
values()All values
d = {"tea": 20, "coffee": 45}
print(list(d.values()))
[20, 45]

Built-in functions

These work without any import. The most useful ones for data work are len, sum, min, max, sorted, round, range, enumerate and zip.

MethodWhat it doesExampleOutput
abs(x)Distance from zero
print(abs(-7.5))
7.5
all(items)True if every item is truthy
print(all([True, 1, "a"]), all([1, 0]))
True False
any(items)True if at least one item is truthy
print(any([0, "", 5]))
True
bool(x)Converts to True or False
print(bool(0), bool("hi"))
False True
chr(n)The character for a number
print(chr(65))
A
dict(...)Makes a dictionary
print(dict(a=1, b=2))
{'a': 1, 'b': 2}
divmod(a, b)Whole groups and remainder together
print(divmod(135, 60))
(2, 15)
enumerate(items)Pairs each item with its position
print(list(enumerate(["a", "b"], start=1)))
[(1, 'a'), (2, 'b')]
filter(f, items)Keeps items where f gives a truthy value
print(list(filter(lambda x: x > 2, [1, 2, 3, 4])))
[3, 4]
float(x)Converts to a decimal number
print(float("3.5"))
3.5
format(x, spec)Formats a value like an f-string slot
print(format(1234.5, ",.2f"))
1,234.50
input(prompt)Asks the user and gives back their text (always a str)
name = "Asha"   # name = input("Name: ") in a real program
print(name)
Asha
int(x)Converts to a whole number (cuts decimals)
print(int("42"), int(3.9))
42 3
isinstance(x, T)True if x is of type T
print(isinstance(5, int))
True
len(x)Number of items or characters
print(len([1, 2, 3]), len("data"))
3 4
list(x)Makes a list
print(list("abc"))
['a', 'b', 'c']
map(f, items)Applies f to every item
print(list(map(str.upper, ["a", "b"])))
['A', 'B']
max(items)Largest item
print(max([4, 9, 1]), max(["kiwi", "apple"], key=len))
9 apple
min(items)Smallest item
print(min([4, 9, 1]))
1
next(it)The next item from an iterator
it = iter([10, 20])
print(next(it), next(it))
10 20
ord(c)The number for a character
print(ord("A"))
65
pow(a, b)a to the power b (same as a ** b)
print(pow(2, 10))
1024
print(...)Shows values; sep goes between, end goes after
print(1, 2, 3, sep="-", end="!\n")
1-2-3!
range(start, stop, step)A run of whole numbers; stop is excluded
print(list(range(2, 11, 4)))
[2, 6, 10]
reversed(items)The items in reverse order
print(list(reversed([1, 2, 3])))
[3, 2, 1]
round(x, n)Rounds to n decimals (.5 goes to the even number)
print(round(3.14159, 2), round(2.5))
3.14 2
set(x)Makes a set (removes duplicates)
print(set([3, 1, 3, 2]))
{1, 2, 3}
sorted(items)A new sorted list
print(sorted([3, 1, 2]), sorted("cab"))
[1, 2, 3] ['a', 'b', 'c']
str(x)Converts to text
print(str(42) + "!")
42!
sum(items)Adds up numbers
print(sum([120, 80, 50]))
250
tuple(x)Makes a tuple
print(tuple([1, 2]))
(1, 2)
type(x)The type of a value
print(type(3.5))
<class 'float'>
zip(a, b)Pairs items from two lists
print(list(zip(["Asha", "Ravi"], [82, 67])))
[('Asha', 82), ('Ravi', 67)]
